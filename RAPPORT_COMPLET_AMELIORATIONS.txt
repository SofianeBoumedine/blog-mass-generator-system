📊 FONCTIONNALITÉS:

1. CACHE EXACT (HASH-BASED)
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   • Correspondance exacte ultra-rapide (O(1))
   • TTL configurable (défaut: 7 jours)
   • Persistance sur disque

   Exemple:
   Prompt: "Écris un article sur plombier paris"
   Hash:   "a3f7c9d2..."
   ➜ Requête identique = Réutilise la réponse cachée


2. CACHE SIMILAIRE (FUZZY MATCHING)
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   Utilise la similarité de Jaccard pour détecter les prompts proches:

   Similarité = Intersection(Mots1, Mots2) / Union(Mots1, Mots2)

   Exemple:
   Prompt A: "article plombier paris 15 urgence"
   Prompt B: "article plombier paris 16 urgence dépannage"
   ➜ Similarité: 4/6 = 66.7%

   Seuil par défaut: 85%
   Si similarité ≥ 85% ➜ Réutilise la réponse existante

   AVANTAGE: Capte les variations mineures (arrondissement, synonymes, etc.)


3. NORMALISATION INTELLIGENTE
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

💻 UTILISATION:

```javascript
const IntelligentCache = require('./lib/intelligentCache');
const cache = new IntelligentCache({
    enabled: true,
    ttl: 7 * 24 * 60 * 60 * 1000,      // 7 jours
    maxSize: 100 * 1024 * 1024,         // 100MB
    similarityThreshold: 0.85           // 85%
});

// Avant un appel API, vérifier le cache
const cachedResponse = await cache.get(prompt, options);
```

```javascript
const IntelligentCache = require('./lib/intelligentCache');
const QualityControl = require('./lib/contentQualityControl');
const PerformanceMonitor = require('./lib/performanceMonitor');

// Initialisation
const cache = new IntelligentCache({ enabled: true });
const qc = new QualityControl({ minScore: 70 });
const monitor = new PerformanceMonitor({ enabled: true });

// Fonction principale de génération
async function generateSite(domain, keywords) {
    console.log(`\n🚀 Génération du site: ${domain}\n`);

    const siteEventId = monitor.startGeneration('site', { domain });

    try {
        // 1. Homepage
        const homepage = await generatePage('homepage', {
            domain,
            keyword: keywords[0]
        });

        // 2. Articles
        const articles = [];
        for (const keyword of keywords) {
            const article = await generatePage('article', {
                domain,
                keyword
            });
            articles.push(article);
        }

        // 3. Blog
        const blog = await generatePage('blog', {
            domain,
            articles: articles.map(a => a.title)
        });

        monitor.endGeneration(siteEventId, true, {
            pages: 2 + articles.length
        });

        // Rapport final
        console.log('\n' + monitor.generateReport());
        console.log(cache.generateReport());

        return { homepage, articles, blog };

    } catch (error) {
        monitor.endGeneration(siteEventId, false);
        monitor.recordError(error, { domain });
        throw error;
    }
}

async function generatePage(type, data) {
    const eventId = monitor.startGeneration(type, data);

    try {
        // 1. Construire le prompt
        const prompt = buildPrompt(type, data);

        // 2. Vérifier le cache
        let content = await cache.get(prompt);

        if (!content) {
            // 3. Appel API avec monitoring
            const start = Date.now();
            content = await aiClient.generate(prompt);
            const duration = Date.now() - start;

            const tokens = Math.ceil((prompt.length + content.length) / 4);
            monitor.recordApiCall('gpt-4', duration, tokens, true);

            // 4. Sauvegarder dans le cache
            await cache.set(prompt, content);
        }

        // 5. Contrôle qualité
        const analysis = await qc.analyzeContent(content, data);
        monitor.recordQualityAnalysis(analysis);

        // 6. Régénérer si qualité insuffisante
        if (analysis.globalScore < 70 && !content.fromCache) {
            console.log('⚠️ Régénération pour améliorer la qualité...');

            const improvedPrompt = prompt + '\n\nATTENTION:\n' +
                analysis.suggestions.join('\n');

            const start = Date.now();
            content = await aiClient.generate(improvedPrompt);
            const duration = Date.now() - start;

            const tokens = Math.ceil((improvedPrompt.length + content.length) / 4);
            monitor.recordApiCall('gpt-4', duration, tokens, true);

            // Ré-analyser
            analysis = await qc.analyzeContent(content, data);
            monitor.recordQualityAnalysis(analysis);
        }

        // 7. Succès
        monitor.endGeneration(eventId, true, {
            score: analysis.globalScore,
            grade: analysis.grade
        });

        console.log(`  ✅ ${type} généré (${analysis.grade}, ${analysis.globalScore}/100)`);

        return { content, analysis };

    } catch (error) {
        monitor.endGeneration(eventId, false);
        monitor.recordError(error, { type, data });
        throw error;
    }
}

// Mettre à jour les stats du cache périodiquement
setInterval(() => {
    monitor.updateCacheStats(cache.getStats());
}, 60 * 1000); // Toutes les minutes

// Export et rapport quotidien
setInterval(async () => {
    console.log('\n' + monitor.generateReport());
    await monitor.exportData();
    await cache.saveCacheStats();
}, 24 * 60 * 60 * 1000); // Toutes les 24h

module.exports = { generateSite, generatePage };
```
